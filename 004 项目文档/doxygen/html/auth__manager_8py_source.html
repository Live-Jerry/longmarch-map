38 def generate_token(self, user):
39 """
40 @brief 为用户生成认证令牌
41 @param user 用户数据字典（包含 id, username, role）
42 @return str Base64 编码的令牌字符串
43 """
44 payload = {
45 "user_id": user["id"],
46 "username": user["username"],
47 "role": user["role"],
48 "exp": int(time.time()) + self.expiration,
49 "iat": int(time.time()),
50 }
51 # 第一部分：payload JSON
52 payload_b64 = base64.urlsafe_b64encode(
53 json.dumps(payload, ensure_ascii=False).encode()
54 ).decode()
55
56 # 第二部分：HMAC 签名
57 sig = hmac.new(
58 self.secret_key,
59 payload_b64.encode(),
60 hashlib.sha256
61 ).hexdigest()
62
63 return f"{payload_b64}.{sig}"
64
65 def verify_token(self, token):
66 """
67 @brief 验证令牌有效性
68 @param token 令牌字符串
69 @return dict|None 成功返回 payload，失败返回 None
70 """
71 try:
72 parts = token.split(".")
73 if len(parts) != 2:
74 return None
75 payload_b64, sig = parts
76
77 # 校验签名
78 expected_sig = hmac.new(
79 self.secret_key,
80 payload_b64.encode(),
81 hashlib.sha256
82 ).hexdigest()
83 if not hmac.compare_digest(sig, expected_sig):
84 return None
85
86 # 解析 payload
87 payload = json.loads(
88 base64.urlsafe_b64decode(payload_b64.encode()).decode()
89 )
90
91 # 检查过期
92 if payload.get("exp", 0) < int(time.time()):
93 return None
94
95 return payload
96
97 except (ValueError, KeyError, json.JSONDecodeError):
98 return None

100 def token_from_request(self, request):
101 """
102 @brief 从 Flask 请求中提取令牌
103 @param request Flask Request 对象
104 @return str|None 令牌字符串
105 """
106 # 优先从 Authorization: Bearer <token> 头获取
107 auth_header = request.headers.get("Authorization", "")
108 if auth_header.startswith("Bearer "):
109 return auth_header[7:]
110
111 # 其次从 URL query 参数获取
112 return request.args.get("token")

114 def current_user(self, request):
115 """
116 @brief 获取当前请求的登录用户
117 @param request Flask Request 对象
118 @return dict|None 用户数据，未登录返回 None
119 """
120 token = self.token_from_request(request)
121 if not token:
122 return None
123 payload = self.verify_token(token)
124 if not payload:
125 return None
126 # 重新从数据库获取最新用户信息
127 return User.get_by_id(payload["user_id"])

133 def register(self, username, password):
134 """
135 @brief 用户注册
136 @param username 用户名
137 @param password 密码
138 @return tuple (user, token) 成功返回用户数据和令牌
139 @raises ValueError 用户名已存在或密码不合规

146 user = User.create(username, password)

160 raise ValueError("用户名或密码错误")
