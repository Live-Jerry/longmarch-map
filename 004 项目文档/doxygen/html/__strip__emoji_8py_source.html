1#!/usr/bin/env python
2# -*- coding: utf-8 -*-
3"""Strip emoji characters from markdown files to prevent GBK encoding issues in CHM."""
4import os, re, glob
5
6emoji_pattern = re.compile(
7 '['
8 '\U0001F300-\U0001F9FF'
9 '\U0001FA00-\U0001FA6F'
10 '\U0001FA70-\U0001FAFF'
11 '\u2600-\u27BF'
12 '\u2702-\u27B0'
13 '\u00A9\u00AE'
14 '\u2122'
15 '\uFE00-\uFE0F'
16 '\u200D'
17 ']+',
18 re.UNICODE
19)
20
21root = r'D:\长征文化'
22files = []
23for pat in ['*.md']:
24 files.extend(glob.glob(os.path.join(root, pat)))
25 files.extend(glob.glob(os.path.join(root, '004 项目文档', '*.md')))
26
27exclude = ['memory', 'IDENTITY', 'SOUL', 'AGENTS', 'USER', 'TOOLS', 'HEARTBEAT', 'MEMORY']
28files = [f for f in files if not any(x in f for x in exclude)]
29
30total_removed = 0
31modified_files = 0
32for fp in sorted(set(files)):
33 with open(fp, 'r', encoding='utf-8') as f:
34 content = f.read()
35 new_content, n = emoji_pattern.subn('', content)
36 if n > 0:
37 with open(fp, 'w', encoding='utf-8') as f:
38 f.write(new_content)
39 total_removed += n
40 modified_files += 1
41 print(f'{os.path.basename(fp)}: removed {n} emoji')
42
43print(f'\nDone: {total_removed} emoji removed from {modified_files} files')
