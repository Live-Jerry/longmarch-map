1#!/usr/bin/env python
2# -*- coding: utf-8 -*-
3"""Find mixed-encoding regions in HHC/HHK files."""
4import os
5
6html_dir = r'D:\长征文化\004 项目文档\doxygen\html'
7
8for fname in ['index.hhc', 'index.hhk']:
9 fp = os.path.join(html_dir, fname)
10 with open(fp, 'rb') as f:
11 raw = f.read()
12
13 print(f'=== {fname} ({len(raw)} bytes) ===')
14
15 # Find regions where UTF-8 multi-byte sequences appear
16 utf8_ranges = []
17 i = 0
18 while i < len(raw):
19 b = raw[i]
20 if b >= 0xC0 and b <= 0xDF:
21 # 2-byte UTF-8
22 if i + 1 < len(raw) and raw[i+1] >= 0x80:
23 utf8_ranges.append((i, i+2))
24 i += 2
25 continue
26 elif b >= 0xE0 and b <= 0xEF:
27 # 3-byte UTF-8
28 if i + 2 < len(raw) and raw[i+1] >= 0x80 and raw[i+2] >= 0x80:
29 utf8_ranges.append((i, i+3))
30 i += 3
31 continue
32 elif b >= 0xF0 and b <= 0xF7:
33 # 4-byte UTF-8
34 if i + 3 < len(raw):
35 utf8_ranges.append((i, i+4))
36 i += 4
37 continue
38 i += 1
39
40 # Merge continuous UTF-8 ranges
41 if utf8_ranges:
42 merged = [utf8_ranges[0]]
43 for start, end in utf8_ranges[1:]:
44 if start <= merged[-1][1] + 1:
45 merged[-1] = (merged[-1][0], max(merged[-1][1], end))
46 else:
47 merged.append((start, end))
48 utf8_ranges = merged
49
50 print(f'UTF-8 multi-byte regions: {len(utf8_ranges)}')
51 for start, end in utf8_ranges:
52 snippet = raw[start:end]
53 try:
54 text = snippet.decode('utf-8')
55 print(f' [{start:06x}-{end:06x}] ({end-start}B) UTF-8: {text[:100]}')
56 except:
57 print(f' [{start:06x}-{end:06x}] ({end-start}B) invalid UTF-8: {snippet[:60].hex()}')
58 print()
