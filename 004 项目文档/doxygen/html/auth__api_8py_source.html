37@auth_bp.route("/register", methods=["POST"])
38def register():
39 """
40 @fn register
41 @brief 用户注册接口
42 @req JSON: { username, password }
43 @res { code, message, data: { user, token } }
44 """
45 data = request.get_json(silent=True) or {}
46 username = data.get("username", "").strip()
47 password = data.get("password", "")
48
49 if not username or not password:
50 return err("用户名和密码不能为空")
51
52 try:
53 manager = AuthManager()
54 user, token = manager.register(username, password)
55 return ok({
56 "user": user,
57 "token": token,
58 })
59 except ValueError as e:
60 return err(str(e))

124@auth_bp.route("/token/verify", methods=["POST"])
125def verify_token():
126 """
127 @fn verify_token
128 @brief 验证令牌有效性
129 @req JSON: { token }
130 @res { code, message, data: payload }
131 """
132 data = request.get_json(silent=True) or {}
133 token = data.get("token", "")
134 if not token:
135 # 尝试从 header 获取
136 auth = request.headers.get("Authorization", "")
137 if auth.startswith("Bearer "):
138 token = auth[7:]
139
140 manager = AuthManager()
141 payload = manager.verify_token(token)
142 if not payload:
143 return err("令牌无效或已过期", code=401)
144 return ok(payload)
