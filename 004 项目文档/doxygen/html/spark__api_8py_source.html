145@spark_bp.route("/<int:spark_id>/review", methods=["PUT"])
146def review_spark(spark_id):
147 """
148 @fn review_spark
149 @brief 审核星火内容（管理员）
150 @param spark_id 星火 ID
151 @req JSON: { status: "approved"|"rejected", admin_comment? }
152 @res { code, data: spark }
153 """
154 try:
155 AuthManager.require_admin(request)
156 except PermissionError as e:
157 return err(str(e), code=403, status=403)
158
159 data = request.get_json(silent=True) or {}
160 status = data.get("status", "")
161 if status not in ("approved", "rejected"):
162 return err("status 必须是 approved 或 rejected")
163
164 try:
165 spark = SparkManager.review_spark(
166 spark_id, status,
167 admin_comment=data.get("admin_comment")
168 )
169 return ok(spark)
170 except ValueError as e:
171 return err(str(e), code=404, status=404)
