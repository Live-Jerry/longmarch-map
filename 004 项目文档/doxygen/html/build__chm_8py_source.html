3"""
4Post-process doxygen output for CHM compatibility.
5Old HTML Help Workshop requires GBK-encoded HTML content.

7Fix: Doxyfile OUTPUT_ENCODING and CHM_INDEX_ENCODING are both UTF-8,
8so Doxygen produces all files (HTML, CSS, JS, HHC, HHK, HHP) in UTF-8.
9This script converts them ALL to GBK for HTML Help Workshop.
10"""

16def convert_file(fp):
17 """Convert one file from UTF-8 to GBK if it contains non-ASCII."""
18 with open(fp, 'rb') as f:
19 raw = f.read()
20
21 # Only process if file has non-ASCII bytes
22 if not any(b > 127 for b in raw):
23 return False
24
25 # Some files already have META charset declaration, some don't.
26 # We decode as UTF-8 (which is what Doxygen now writes), then re-encode as GBK.
27 try:
28 text = raw.decode('utf-8')
29 except UnicodeDecodeError:
30 # Try GBK as fallback
31 try:
32 text = raw.decode('gbk')
33 except:
34 return False # Can't handle this file
35
36 # Update charset declarations in HTML files
37 ext = os.path.splitext(fp)[1].lower()
38 if ext == '.html':
39 text = text.replace('charset=UTF-8', 'charset=GB2312')
40 text = text.replace('charset=utf-8', 'charset=GB2312')
41 elif ext == '.hhc':
42 # HHC files: also add/update charset param if missing
43 # HTML Help Workshop needs charset=gb2312 in the HTML header
44 if 'charset=' not in text[:500].lower():
45 # Insert charset after <HEAD>
46 text = text.replace('<HEAD>', '<HEAD><meta http-equiv="Content-Type" content="text/html; charset=GB2312">')
47 else:
48 text = text.replace('charset=UTF-8', 'charset=GB2312')
49 text = text.replace('charset=utf-8', 'charset=GB2312')
50 elif ext == '.hhk':
51 if 'charset=' not in text[:500].lower():
52 text = text.replace('<HEAD>', '<HEAD><meta http-equiv="Content-Type" content="text/html; charset=GB2312">')
53 else:
54 text = text.replace('charset=UTF-8', 'charset=GB2312')
55 text = text.replace('charset=utf-8', 'charset=GB2312')
56
57 # Encode as GBK (superset of GB2312, handles more Chinese chars)
58 try:
59 gbk = text.encode('gbk')
60 except UnicodeEncodeError as e:
61 # Find and report the specific bad chars
62 bad_chars = set()
63 for i in range(e.start, min(e.start + 100, len(text))):
64 try:
65 text[i].encode('gbk')
66 except:
67 cp = ord(text[i])
68 bad_chars.add(f'U+{cp:04X}')
69 print(f' WARN: {os.path.basename(fp)} has non-GBK chars: {bad_chars}')
70 # Strip non-GBK characters instead of replacing with ?
71 clean = ''
72 for c in text:
73 try:
74 c.encode('gbk')
75 clean += c
76 except:
77 pass # strip it entirely
78 gbk = clean.encode('gbk')
79
80 with open(fp, 'wb') as f:
81 f.write(gbk)
82 return True

84def convert_to_gbk(directory):
85 """Convert all relevant files from UTF-8 to GBK."""
86 count = 0
87 # Files to convert: HTML, CSS, JS, HHC, HHK, HHP
88 patterns = ['*.html', '*.css', '*.js', '*.hhc', '*.hhk', '*.hhp']
89 for pattern in patterns:
90 for fp in glob.glob(os.path.join(directory, pattern)):
91 if convert_file(fp):
92 count += 1
93 if count % 50 == 0:
94 print(f' ... converted {count} files')
95 return count
