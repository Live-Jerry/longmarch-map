46 @staticmethod
47 def save_file(file_obj, media_type):
48 """
49 @brief 保存上传的文件并返回相对路径
50 @param file_obj Flask FileStorage 对象
51 @param media_type 素材类型（image/audio/video/text）
52 @return str 保存后的相对路径
53 @raises ValueError 文件类型不支持
54 """
55 if not file_obj or not file_obj.filename:
56 raise ValueError("未提供文件")
57
58 # 检查 MIME 类型
59 mime = file_obj.content_type or "application/octet-stream"
60 allowed = MediaManager.ALLOWED_MIME.get(media_type, [])
61 if allowed and mime not in allowed:
62 raise ValueError(f"不支持的文件类型: {mime}，允许: {allowed}")
63
64 # 生成唯一文件名
65 ext = os.path.splitext(file_obj.filename)[1].lower()
66 unique_name = f"{uuid.uuid4().hex}{ext}"
67
68 upload_dir = MediaManager.get_upload_dir(media_type)
69 file_path = os.path.join(upload_dir, unique_name)
70
71 file_obj.save(file_path)
72
73 # 返回相对路径（相对于 static/uploads）
74 return os.path.join("uploads", media_type, unique_name)
