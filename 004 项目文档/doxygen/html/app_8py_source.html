199def _ensure_admin(cursor):
200 """插入默认管理员账户（若不存在）"""
201 cursor.execute("SELECT id FROM user WHERE username = ?", (config.Config.ADMIN_USERNAME,))
202 if not cursor.fetchone():
203 pw_hash = hashlib.sha256(config.Config.ADMIN_PASSWORD.encode()).hexdigest()
204 now = datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%SZ")
205 cursor.execute(
206 "INSERT INTO user (username, password_hash, role, created_at) VALUES (?, ?, ?, ?)",
207 (config.Config.ADMIN_USERNAME, pw_hash, "admin", now)
208 )
