1#!/usr/bin/env python
2"""Check HHP file for missing referenced files."""
3import os, sys
4sys.stdout.reconfigure(encoding='utf-8', errors='replace')
5
6html_dir = r'D:\长征文化\004 项目文档\doxygen\html'
7
8with open(os.path.join(html_dir, 'index.hhp'), 'rb') as f:
9 raw = f.read()
10text = raw.decode('gbk')
11
12# Find [FILES] section
13files_start = text.find('[FILES]')
14files_end = text.find('\n\n', files_start)
15if files_end == -1:
16 files_end = len(text)
17
18files_section = text[files_start:files_end]
19lines = files_section.split('\n')[1:] # skip [FILES]
20files = [f.strip() for f in lines if f.strip()]
21
22missing = []
23for fname in files:
24 if not os.path.exists(os.path.join(html_dir, fname)):
25 missing.append(fname)
26
27print(f'Total files listed: {len(files)}')
28print(f'Missing files: {len(missing)}')
29if missing:
30 print('Missing:')
31 for f in missing[:20]:
32 print(f' {f}')
33 if len(missing) > 20:
34 print(f' ... and {len(missing)-20} more')
