1#!/usr/bin/env python
2"""Check encoding of Doxygen-generated HHC/HHK before build_chm.py conversion."""
3import os, re, sys
4sys.stdout.reconfigure(encoding='utf-8', errors='replace')
5
6html_dir = r'D:\长征文化\004 项目文档\doxygen\html'
7
8for fname in ['index.hhc', 'index.hhk']:
9 fp = os.path.join(html_dir, fname)
10 with open(fp, 'rb') as f:
11 raw = f.read()
12
13 # Try UTF-8
14 try:
15 t_utf8 = raw.decode('utf-8')
16 utf8_ok = True
17 except:
18 utf8_ok = False
19
20 # Try GBK
21 try:
22 t_gbk = raw.decode('gbk')
23 gbk_ok = True
24 except:
25 gbk_ok = False
26
27 print(f'{fname} ({len(raw)} bytes)')
28 print(f' Decodes as UTF-8: {utf8_ok}')
29 print(f' Decodes as GBK: {gbk_ok}')
30
31 if utf8_ok and gbk_ok:
32 print(f' => Valid in BOTH encodings (ambiguous)')
33 elif utf8_ok and not gbk_ok:
34 print(f' => UTF-8 only (GBK decode fails)')
35 elif gbk_ok and not utf8_ok:
36 print(f' => GBK only (UTF-8 decode fails)')
37 else:
38 print(f' => Neither encoding works!')
39
40 # Check for Name param values
41 names_gbk = re.findall(r'param\s+name="Name"\s+value="([^"]*)"', t_gbk) if gbk_ok else []
42 names_utf8 = re.findall(r'param\s+name="Name"\s+value="([^"]*)"', t_utf8) if utf8_ok else []
43
44 if gbk_ok and names_gbk:
45 print(f' GBK names ({len(names_gbk)}): {names_gbk[0][:60]}...')
46 if utf8_ok and names_utf8:
47 print(f' UTF-8 names ({len(names_utf8)}): {names_utf8[0][:60]}...')
48
49 # Check for raw UTF-8 sequences
50 utf8_bytes = 0
51 for b in raw:
52 if b >= 0x80:
53 utf8_bytes += 1
54 print(f' Non-ASCII bytes: {utf8_bytes} ({utf8_bytes/len(raw)*100:.1f}%)')
55
56 # Check for UTF-8 multi-byte patterns that indicate UTF-8 content
57 utf8_seq = 0
58 for i in range(len(raw) - 2):
59 if raw[i] >= 0xE0 and raw[i] <= 0xEF and raw[i+1] >= 0x80 and raw[i+2] >= 0x80:
60 utf8_seq += 1
61 print(f' UTF-8 3-byte sequences: {utf8_seq}')
62 print()
