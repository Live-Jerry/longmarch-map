84 """
85 @fn upload_media

87 @req multipart/form-data: file, type, title, description, node_id
88 @res { code, data: media }
89 """
90 user = _require_login()
91 if isinstance(user, tuple):
92 return user
93
94 if "file" not in request.files:
95 return err("未上传文件")
96
97 file_obj = request.files["file"]
98 media_type = request.form.get("type", "image")
99 title = request.form.get("title", file_obj.filename or "未命名")
100 description = request.form.get("description", "")
101 node_id = request.form.get("node_id", "")
102
103 data = {
104 "type": media_type,
105 "title": title,
106 "description": description,
107 "node_id": node_id,
108 "uploader_id": user["id"],
109 }
110
111 try:
112 media = MediaManager.create_media(data, file_obj)
113 return ok(media), 201
114 except ValueError as e:
115 return err(str(e))

130 admin = _require_admin()

133
134 data = request.get_json(silent=True) or {}
