85 @classmethod
86 def get_all(cls, page=1, per_page=20, search=None, province=None, status=None):
87 """
88 @brief 分页获取节点列表
89 @param page 页码（从 1 开始）
90 @param per_page 每页数量
91 @param search 模糊搜索关键词（搜索 title / location）
92 @param province 省份过滤
93 @param status 状态过滤，默认只返回 active
94 @return tuple (节点列表, 总数)
95 """
96 conn = _get_db()
97 conditions = []
98 params = []
99
100 if status:
101 conditions.append("status = ?")
102 params.append(status)
103 else:
104 conditions.append("status = 'active'")
105
106 if search:
107 conditions.append("(title LIKE ? OR location LIKE ?)")
108 params.extend([f"%{search}%", f"%{search}%"])
109
110 if province:
111 conditions.append("location LIKE ?")
112 params.append(f"%{province}%")
113
114 where_clause = " AND ".join(conditions) if conditions else "1=1"
115
116 total = conn.execute(
117 f"SELECT COUNT(*) FROM node WHERE {where_clause}", params
118 ).fetchone()[0]
119
120 offset = (page - 1) * per_page
121 rows = conn.execute(
122 f"""SELECT * FROM node
123 WHERE {where_clause}
124 ORDER BY time ASC
125 LIMIT ? OFFSET ?""",
126 params + [per_page, offset]
127 ).fetchall()
128
129 return [cls.to_row_dict(r) for r in rows], total

155 @classmethod
156 def get_nearby(cls, lat, lng, radius_km=50):
157 """
158 @brief 获取指定坐标附近的所有节点
159 @param lat 纬度
160 @param lng 经度
161 @param radius_km 搜索半径（公里），默认 50km
162 @note 使用简化球面距离公式
163 @return list 附近节点列表（按距离升序）
164 """
165 lat_delta = radius_km / 111.0
166 lng_delta = radius_km / (111.0 * max(0.01, abs(lat)))
167
168 rows = _get_db().execute(
169 """SELECT * FROM node
170 WHERE lat BETWEEN ? AND ?
171 AND lng BETWEEN ? AND ?
172 AND status = 'active'
173 ORDER BY ABS(lat - ?) + ABS(lng - ?) ASC""",
174 (lat - lat_delta, lat + lat_delta,
175 lng - lng_delta, lng + lng_delta,
176 lat, lng)
177 ).fetchall()
178 return [cls.to_row_dict(r) for r in rows]

184 @classmethod
185 def create(cls, data):
186 """
187 @brief 创建新节点
188 @param data 节点数据字典（包含 node_id, title, location, lat, lng 等）
189 @return dict 新创建的节点数据
190 """
191 now = datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%SZ")
192
193 if "image_list" in data and isinstance(data["image_list"], list):
194 data["image_list"] = json.dumps(data["image_list"], ensure_ascii=False)
195
196 fields = {k: v for k, v in data.items() if k in cls.ALL_FIELDS}
197 fields.setdefault("status", "active")
198 fields.setdefault("created_at", now)
199 fields.setdefault("updated_at", now)
200
201 columns = ", ".join(fields.keys())
202 placeholders = ", ".join(["?"] * len(fields))
203 values = list(fields.values())
204
205 conn = _get_db()
206 cursor = conn.execute(
207 f"INSERT INTO node ({columns}) VALUES ({placeholders})", values
208 )
209 _commit(conn)
210 return cls.get_by_node_id(data["node_id"])
