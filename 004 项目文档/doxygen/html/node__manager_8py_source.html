122 @staticmethod
123 def import_from_json(json_path=None):
124 """
125 @brief 从 JSON 文件批量导入节点初始数据
126 @param json_path JSON 文件路径（默认使用配置的初始数据路径）
127 @return dict 导入结果统计
128 """
129 if json_path is None:
130 json_path = config.NODE_DATA_JSON
131
132 if not os.path.exists(json_path):
133 raise FileNotFoundError(f"节点数据文件不存在: {json_path}")
134
135 with open(json_path, "r", encoding="utf-8") as f:
136 nodes_data = json.load(f)
137
138 imported = 0
139 skipped = 0
140 errors = []
141
142 for item in nodes_data:
143 try:
144 node_id = item.get("node_id")
145 if not node_id:
146 errors.append(f"缺少 node_id: {item}")
147 skipped += 1
148 continue
149
150 existing = Node.get_by_node_id(node_id)
151 if existing:
152 skipped += 1
153 continue
154
155 Node.create(item)
156 imported += 1
157 except Exception as e:
158 errors.append(f"导入失败 {item.get('node_id', '?')}: {str(e)}")
159 skipped += 1
160
161 return {
162 "imported": imported,
163 "skipped": skipped,
164 "errors": errors,
165 }
