79 @classmethod
80 def get_all(cls, page=1, per_page=20, media_type=None, status=None):
81 """@brief 分页获取素材列表"""
82 conn = _get_db()
83 conditions = []
84 params = []
85
86 if media_type:
87 conditions.append("type = ?")
88 params.append(media_type)
89 if status:
90 conditions.append("status = ?")
91 params.append(status)
92
93 where = " AND ".join(conditions) if conditions else "1=1"
94
95 total = conn.execute(
96 f"SELECT COUNT(*) FROM media WHERE {where}", params
97 ).fetchone()[0]
98
99 offset = (page - 1) * per_page
100 rows = conn.execute(
101 f"SELECT * FROM media WHERE {where} ORDER BY created_at DESC LIMIT ? OFFSET ?",
102 params + [per_page, offset]
103 ).fetchall()
104 return [cls.to_dict(r) for r in rows], total

110 @classmethod
111 def create(cls, data):
112 """@brief 创建素材记录"""
113 now = datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%SZ")
114 data.setdefault("status", "pending")
115 data.setdefault("created_at", now)
116
117 valid_fields = {
118 "node_id", "type", "title", "description",
119 "file_path", "source", "uploader_id", "status", "created_at"
120 }
121 fields = {k: v for k, v in data.items() if k in valid_fields}
122
123 columns = ", ".join(fields.keys())
124 placeholders = ", ".join(["?"] * len(fields))
125
126 conn = _get_db()
127 cursor = conn.execute(
128 f"INSERT INTO media ({columns}) VALUES ({placeholders})",
129 list(fields.values())
130 )
131 _commit(conn)
132 return cls.get_by_id(cursor.lastrowid)

134 @classmethod
135 def update_status(cls, media_id, status, admin_comment=None):
