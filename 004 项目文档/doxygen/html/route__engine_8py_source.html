138 """
139 @brief 计算自动漫游路径（带时间估算）
140 @param army 军队编号
141 @param speed_kmh 假设移动速度（公里/小时），默认 30 km/h（乘车）
142 @return dict 路径数据，含每个节点的时间估算
143 """
144 route = RouteEngine.get_route_by_army(army)
145 points = route["points"]
146
147 if not points:
148 return {"army": army, "segments": [], "total_km": 0, "total_hours": 0}
149
150 import math
151 total_km = 0.0
152 segments = []
153
154 for i, point in enumerate(points):
155 if i == 0:
156 seg = {
157 "index": i,
158 "node_id": point.get("node_id"),
159 "title": point.get("node_title") or point.get("location", ""),
160 "lat": point["lat"],
161 "lng": point["lng"],
162 "dist_km": 0,
163 "duration_h": 0,

165 else:
166 prev = points[i - 1]
167 dist = RouteEngine._haversine_km(
168 prev["lat"], prev["lng"],
169 point["lat"], point["lng"]
170 )
171 total_km += dist
172 duration = dist / speed_kmh
173
174 seg = {
175 "index": i,
176 "node_id": point.get("node_id"),
177 "title": point.get("node_title") or point.get("location", ""),
178 "lat": point["lat"],
179 "lng": point["lng"],
180 "dist_km": round(dist, 2),
181 "duration_h": round(duration, 2),

183 segments.append(seg)
184
185 total_hours = total_km / speed_kmh
186
187 return {

196 @staticmethod
197 def _haversine_km(lat1, lng1, lat2, lng2):
198 """
199 @brief 计算两点间的球面距离（公里）
200 @param lat1, lng1 第一个点的纬度和经度
201 @param lat2, lng2 第二个点的纬度和经度
202 @return float 距离（公里）
203 """
204 import math
205 R = 6371.0 # 地球半径（公里）
206
207 phi1 = math.radians(lat1)
208 phi2 = math.radians(lat2)
209 dphi = math.radians(lat2 - lat1)
210 dlam = math.radians(lng2 - lng1)
211
212 a = math.sin(dphi / 2) ** 2 + \
213 math.cos(phi1) * math.cos(phi2) * math.sin(dlam / 2) ** 2
214 c = 2 * math.atan2(math.sqrt(a), math.sqrt(1 - a))
215
216 return R * c
