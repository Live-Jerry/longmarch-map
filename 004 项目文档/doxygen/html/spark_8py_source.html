59 @classmethod
60 def get_all(cls, page=1, per_page=20, status=None, node_id=None):
61 """@brief 分页获取星火列表"""
62 conn = _get_db()
63 conditions = []
64 params = []
65
66 if status:
67 conditions.append("s.status = ?")
68 params.append(status)
69 else:
70 conditions.append("s.status = 'approved'")
71
72 if node_id:
73 conditions.append("s.node_id = ?")
74 params.append(node_id)
75
76 where = " AND ".join(conditions) if conditions else "1=1"
77
78 total = conn.execute(
79 f"SELECT COUNT(*) FROM spark s WHERE {where}", params
80 ).fetchone()[0]
81
82 offset = (page - 1) * per_page
83 rows = conn.execute(
84 f"""SELECT s.*, u.username
85 FROM spark s
86 LEFT JOIN user u ON s.user_id = u.id
87 WHERE {where}
88 ORDER BY s.created_at DESC
89 LIMIT ? OFFSET ?""",
90 params + [per_page, offset]
91 ).fetchall()
92 return [cls.to_dict(r) for r in rows], total

143 @classmethod
144 def create(cls, data):
145 """@brief 提交新的星火内容"""
146 now = datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%SZ")
147 data.setdefault("status", "pending")
148 data.setdefault("created_at", now)
149
150 valid_fields = {
151 "node_id", "user_id", "title", "content", "media_type",
152 "file_path", "source", "status", "created_at"
153 }
154 fields = {k: v for k, v in data.items() if k in valid_fields}
155
156 columns = ", ".join(fields.keys())
157 placeholders = ", ".join(["?"] * len(fields))
158
159 conn = _get_db()
160 cursor = conn.execute(
161 f"INSERT INTO spark ({columns}) VALUES ({placeholders})",
162 list(fields.values())
163 )
164 _commit(conn)
165 return cls.get_by_id(cursor.lastrowid)

167 @classmethod
168 def review(cls, spark_id, status, admin_comment=None):
169 """@brief 管理员审核星火内容"""
170 if status not in ("approved", "rejected"):
171 raise ValueError("审核状态必须为 approved 或 rejected")
172
173 conn = _get_db()
174 if admin_comment:
175 conn.execute(
176 "UPDATE spark SET status = ?, admin_comment = ? WHERE id = ?",
177 (status, admin_comment, spark_id)
178 )
179 else:
180 conn.execute(
181 "UPDATE spark SET status = ? WHERE id = ?",
182 (status, spark_id)
183 )
184 _commit(conn)
185 return cls.get_by_id(spark_id)
